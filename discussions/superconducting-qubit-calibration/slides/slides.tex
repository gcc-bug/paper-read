\begin{frame}{T1 测衰减；Ramsey 测相位保持}
 \begin{columns}[T,onlytextwidth]
 \column{.47\textwidth}
 \textbf{T1}\par
 $\pi\ \to\ \mathrm{delay}\ t\ \to\ \mathrm{readout}$
 \[P_1(t)=B+A e^{-t/T_1}\]
 相对 baseline 的 excess\par
 在 $T_1$ 后剩下 $1/e$。
 \column{.50\textwidth}
 \textbf{Ramsey}\par
 $\pi/2\ \to\ t\ \to\ \pi/2\ \to\ \mathrm{readout}$
 \[P_1(t)=B+A e^{-t/T_2^*}\cos(2\pi f_Rt+\phi)\]
 第二个 pulse 将 phase\par
 转成 population contrast。
 \end{columns}
 \src{Stationary exponential 模型；$f_R$ 为 fringe frequency。Krantz Fig. 5 \cite{krantz}。}
\end{frame}

\end{frame}

\begin{frame}{校准参数需要六类检查共同支持}
 \begin{columns}[T,onlytextwidth]
 \column{.47\textwidth}
 \textbf{Observable}\par 记录的量与模型是否一致？\par\vspace{2mm}

 \textbf{Reproducibility}\par 重复与改变 sweep 后是否一致？
 \end{columns}

\end{frame}
